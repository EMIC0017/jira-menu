# JiraMenu Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A minimal macOS menubar app that lets a user search Jira Cloud issues as they type, opens them in the browser, and can be distributed via a personal Homebrew tap.

**Architecture:** Single-binary SwiftUI app targeting macOS 13+, built as an Xcode project. A `MenuBarExtra` in the status bar reveals a popover with a search field and live results. Credentials are stored in the macOS Keychain. An `InputParser` maps user text to a JQL query; results come from Jira Cloud's REST v3 search API. Distribution is an unsigned `.dmg` via GitHub Releases, installed through a personal Homebrew cask.

**Tech Stack:** Swift 6.3, SwiftUI (macOS 13 Ventura minimum, uses `MenuBarExtra`), XCTest, URLSession, `Security.framework` (Keychain), **Swift Package Manager** (no Xcode), `hdiutil`, GitHub Actions, Homebrew Cask.

**Why this matters:** When free-text results come back from Jira, we re-rank them client-side so the best matches surface first. The scoring function is small but shapes how the app *feels*. Too strict: good results get buried. Too loose: noise floats up.

**Design decisions you're making:**
- Does exact key match always win? (Probably yes.)
- Does prefix match beat contains match? (Probably yes.)
- Case sensitive? (Probably no.)
- Bonus for matches at word boundaries? (Probably yes.)

```swift
import Foundation
import Security

struct KeychainStore {
    let service: String

    enum KeychainError: Error { case unexpectedStatus(OSStatus) }

    init(service: String = "dev.ericmorin.jiramenu") {
        self.service = service
    }

    func save(_ credentials: Credentials) throws {
        let payload: [String: String] = [
            "siteURL": credentials.siteURL.absoluteString,
            "email": credentials.email,
            "apiToken": credentials.apiToken,
        ]
        let data = try JSONEncoder().encode(payload)

        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
        ]
        SecItemDelete(query as CFDictionary)

        var attrs = query
        attrs[kSecValueData as String] = data
        let status = SecItemAdd(attrs as CFDictionary, nil)
        guard status == errSecSuccess else { throw KeychainError.unexpectedStatus(status) }
    }

    func load() throws -> Credentials? {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne,
        ]
        var result: AnyObject?
        let status = SecItemCopyMatching(query as CFDictionary, &result)
        if status == errSecItemNotFound { return nil }
        guard status == errSecSuccess, let data = result as? Data else {
            throw KeychainError.unexpectedStatus(status)
        }
        let payload = try JSONDecoder().decode([String: String].self, from: data)
        guard
            let urlString = payload["siteURL"],
            let url = URL(string: urlString),
            let email = payload["email"],
            let token = payload["apiToken"]
        else { return nil }
        return Credentials(siteURL: url, email: email, apiToken: token)
    }

    func delete() throws {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
        ]
        let status = SecItemDelete(query as CFDictionary)
        guard status == errSecSuccess || status == errSecItemNotFound else {
            throw KeychainError.unexpectedStatus(status)
        }
    }
}
```

```swift
import Foundation

final class MockURLProtocol: URLProtocol {
    static var handler: ((URLRequest) throws -> (HTTPURLResponse, Data))?

    override class func canInit(with request: URLRequest) -> Bool { true }
    override class func canonicalRequest(for request: URLRequest) -> URLRequest { request }

    override func startLoading() {
        guard let handler = MockURLProtocol.handler else {
            fatalError("MockURLProtocol.handler not set")
        }
        do {
            let (response, data) = try handler(request)
            client?.urlProtocol(self, didReceive: response, cacheStoragePolicy: .notAllowed)
            client?.urlProtocol(self, didLoad: data)
            client?.urlProtocolDidFinishLoading(self)
        } catch {
            client?.urlProtocol(self, didFailWithError: error)
        }
    }

    override func stopLoading() {}
}
```

```swift
        let decoded = try JSONDecoder().decode(SearchResponse.self, from: data)
```

## Task 7: SearchViewModel

Coordinates state between the search field and the results list. Debounces input, cancels in-flight requests, and publishes ranked results.

**Files:**
- Create: `JiraMenu/Core/SearchViewModel.swift`

- [ ] **Step 1: Implement `SearchViewModel`**

```swift
import Foundation
import Combine

@MainActor
final class SearchViewModel: ObservableObject {
    @Published var query: String = ""
    @Published private(set) var results: [Issue] = []
    @Published private(set) var isSearching: Bool = false
    @Published private(set) var errorMessage: String?

    private let client: JiraClient
    private var currentTask: Task<Void, Never>?
    private var cancellables = Set<AnyCancellable>()

    init(client: JiraClient) {
        self.client = client
        $query
            .removeDuplicates()
            .debounce(for: .milliseconds(200), scheduler: DispatchQueue.main)
            .sink { [weak self] q in self?.performSearch(q) }
            .store(in: &cancellables)
    }

    private func performSearch(_ raw: String) {
        currentTask?.cancel()
        let parsed = InputParser.parse(raw)
        if case .empty = parsed {
            results = []
            isSearching = false
            errorMessage = nil
            return
        }
        isSearching = true
        errorMessage = nil
        currentTask = Task { [weak self] in
            guard let self else { return }
            do {
                let fetched = try await self.client.search(parsed)
                if Task.isCancelled { return }
                let ranked: [Issue]
                if case .freeText(let text) = parsed {
                    ranked = FuzzyScorer.rank(query: text, issues: fetched)
                } else {
                    ranked = fetched
                }
                self.results = ranked
                self.isSearching = false
            } catch {
                if Task.isCancelled { return }
                self.errorMessage = error.localizedDescription
                self.results = []
                self.isSearching = false
            }
        }
    }
}
```

```swift
import SwiftUI

struct SettingsView: View {
    @State private var siteURL: String = ""
    @State private var email: String = ""
    @State private var apiToken: String = ""
    @State private var statusMessage: String?
    @State private var isValidating = false

    let onSaved: (Credentials) -> Void

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            Text("Jira Menu — Settings").font(.headline)

            TextField("Site URL (https://acme.atlassian.net)", text: $siteURL)
                .textFieldStyle(.roundedBorder)
            TextField("Email", text: $email)
                .textFieldStyle(.roundedBorder)
            SecureField("API token", text: $apiToken)
                .textFieldStyle(.roundedBorder)

            Link("Get an API token",
                 destination: URL(string: "https://id.atlassian.com/manage-profile/security/api-tokens")!)
                .font(.caption)

            HStack {
                Button(isValidating ? "Checking…" : "Save") {
                    Task { await validateAndSave() }
                }
                .disabled(isValidating || siteURL.isEmpty || email.isEmpty || apiToken.isEmpty)
                if let status = statusMessage {
                    Text(status).font(.caption).foregroundStyle(.secondary)
                }
            }
        }
        .padding()
        .frame(width: 360)
    }

    private func validateAndSave() async {
        guard let url = URL(string: siteURL) else {
            statusMessage = "Invalid URL"
            return
        }
        let creds = Credentials(siteURL: url, email: email, apiToken: apiToken)
        isValidating = true
        defer { isValidating = false }

        var request = URLRequest(url: url.appendingPathComponent("/rest/api/3/myself"))
        request.setValue(creds.basicAuthHeader, forHTTPHeaderField: "Authorization")
        request.setValue("application/json", forHTTPHeaderField: "Accept")
        do {
            let (_, response) = try await URLSession.shared.data(for: request)
            guard let http = response as? HTTPURLResponse, http.statusCode == 200 else {
                statusMessage = "Invalid credentials"
                return
            }
            try KeychainStore().save(creds)
            statusMessage = "Saved ✓"
            onSaved(creds)
        } catch {
            statusMessage = "Failed: \(error.localizedDescription)"
        }
    }
}
```

## Task 11: DMG build script

**Files:**
- Create: `scripts/build-dmg.sh`

Uses `xcodebuild archive` + `hdiutil` — no external tooling required.

- [ ] **Step 1: Create the script**

```bash
#!/usr/bin/env bash
set -euo pipefail

VERSION="${1:?usage: build-dmg.sh <version> (e.g. 0.1.0)}"
APP_NAME="JiraMenu"
BUILD_DIR="build"
ARCHIVE_PATH="${BUILD_DIR}/${APP_NAME}.xcarchive"
EXPORT_DIR="${BUILD_DIR}/export"
DIST_DIR="dist"
DMG_PATH="${DIST_DIR}/${APP_NAME}-${VERSION}.dmg"

rm -rf "${BUILD_DIR}" "${EXPORT_DIR}" "${DIST_DIR}"
mkdir -p "${BUILD_DIR}" "${DIST_DIR}"

xcodebuild archive \
  -project "${APP_NAME}.xcodeproj" \
  -scheme "${APP_NAME}" \
  -archivePath "${ARCHIVE_PATH}" \
  -configuration Release \
  ARCHS="arm64 x86_64" \
  ONLY_ACTIVE_ARCH=NO \
  CODE_SIGN_IDENTITY="-" \
  CODE_SIGNING_REQUIRED=NO

# Copy .app from archive
mkdir -p "${EXPORT_DIR}"
cp -R "${ARCHIVE_PATH}/Products/Applications/${APP_NAME}.app" "${EXPORT_DIR}/"

# Ad-hoc sign so macOS can launch it after clearing quarantine
codesign --force --deep --sign - "${EXPORT_DIR}/${APP_NAME}.app"

# Stage DMG contents with a symlink to /Applications for drag-install
STAGE="${BUILD_DIR}/dmg"
mkdir -p "${STAGE}"
cp -R "${EXPORT_DIR}/${APP_NAME}.app" "${STAGE}/"
ln -s /Applications "${STAGE}/Applications"

hdiutil create -volname "${APP_NAME}" \
  -srcfolder "${STAGE}" \
  -ov -format UDZO \
  "${DMG_PATH}"

echo ""
echo "Built: ${DMG_PATH}"
shasum -a 256 "${DMG_PATH}"
```
